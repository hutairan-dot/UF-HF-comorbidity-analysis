#!/usr/bin/env python3
# ==============================================================================
# Script: 10c_celloracle.py
# Purpose: CellOracle GRN inference (KNN + ridge) and virtual perturbation of
#          HIF1A (UF) / RUNX1, RUNX2 (HF) in Scissor+ fibroblasts; Random KO
#          as negative control. UMAP flow field comparison.
# Expected: KO HIF1A (UF) and KO RUNX1/RUNX2 (HF) alter state transition vectors.
# ==============================================================================
import os, numpy as np, pandas as pd, scanpy as sc, celloracle as co
from celloracle import motif_analysis as ma
from celloracle.applications import Oracle

CONFIG = {
    "uf_adata": "UF_scissor_fib.h5ad",
    "hf_adata": "HF_scissor_fib.h5ad",
    "base_grn": "human_base_GRN.celloracle.baseGRN",  # from celloracle resources
    "ko_uf": ["HIF1A", "RANDOM"],
    "ko_hf": ["RUNX1", "RUNX2", "RANDOM"],
    "out_dir": "celloracle_out",
    "n_grn_units": 2000,
    "seed": 1234
}
os.makedirs(CONFIG["out_dir"], exist_ok=True)

def run_perturbation(adata_path, ko_genes, tag):
    adata = sc.read_h5ad(adata_path)
    sc.pp.normalize_total(adata, target_sum=1e4); sc.pp.log1p(adata)
    sc.pp.highly_variable_genes(adata, n_top_genes=3000)
    sc.tl.pca(adata); sc.pp.neighbors(adata); sc.tl.umap(adata)

    # Load base GRN
    base_grn = co.data.load_human_promoter_base_GRN()  # or load from file

    # Initialize Oracle and infer GRN
    oracle = Oracle(adata, base_grn=base_grn, n_jobs=8, seed=CONFIG["seed"])
    oracle.fit_GRN_for_simulation(use_cluster_label=False)

    for gene in ko_genes:
        if gene == "RANDOM":
            rng = np.random.default_rng(CONFIG["seed"])
            gene = rng.choice(adata.var_names[adata.var["highly_variable"]].tolist())
            label = "RandomKO"
        else:
            label = gene

        oracle.simulate_shift(perturb_condition={gene: 0.0},
                              n_propagation=3)
        oracle.estimate_transition_prob(n_neighbors=200, knn_random=True,
                                         sampled_fraction=1)
        oracle.calculate_embedding_shift(sigma_corr=0.05)

        # Save flow field plot data
        oracle.plot_quiver(scale=1, save=f"{CONFIG['out_dir']}/{tag}_KO_{label}_quiver.pdf")
        oracle.plot_quiver(scale=1, save=f"{CONFIG['out_dir']}/{tag}_KO_{label}_stream.pdf",
                           streamplot=True)
        print(f"[{tag}] KO {label} done.")

if __name__ == "__main__":
    for tag, adata, kos in [("UF", CONFIG["uf_adata"], CONFIG["ko_uf"]),
                             ("HF", CONFIG["hf_adata"], CONFIG["ko_hf"])]:
        run_perturbation(adata, kos, tag)
